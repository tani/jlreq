% Punctuation position in vertical writing via bxcjkvert: measure ink

% the uplatex tate ground truth (ujarticle[tate]) rendered separately.

% of the page (the CJKvert reading model); true vertical typesetting

\usepackage[utf8]{inputenc}

\usepackage[ipaex-type1]{bxcjkjatype}
\usepackage{bxcjkvert}
\makeatletter
\def\markpos#1{\pdfsavepos\write\@auxout{\string\gdef\string#1{\the\pdflastxpos,\the\pdflastypos}}}
\pagestyle{empty}
\begin{document}
\begin{CJK}{UTF8}{min}
\CJKvert
\noindent
\markpos\cellA あ%
\markpos\cellB 、%
\markpos\cellC 。%
\markpos\cellD 「%
\markpos\cellE 」%
\markpos\cellF
\par
\typeout{V3-PUNCT: done}

\end{document}
